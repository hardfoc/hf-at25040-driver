102 const Status first = ReadStatus(a);

173 Status WaitReady(uint32_t timeout_ms = kWriteCycleTimeoutMs) noexcept {
174 constexpr bool kHasDelay = detail::HasDelayUs<Spi>::value;
175 const uint32_t polls = kHasDelay ? (timeout_ms * 1000u) / kPollIntervalUs + 1u
176 : timeout_ms * kPollsPerMsWithoutDelay + 1u;
177 for (uint32_t i = 0; i < polls; ++i) {
178 uint8_t sr = 0xFF;

181 return st; // BusError

183 if ((sr & kSrWip) == 0u) {
184 return Status::Ok;
185 }
186 if constexpr (kHasDelay) {
187 if (i + 1u < polls) {
